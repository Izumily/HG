// Fisher-Yates 洗牌算法，打乱选项顺序
function shuffleArray(arr) {
    let newArr = [...arr];
    for (let i = newArr.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [newArr[i], newArr[j]] = [newArr[j], newArr[i]];
    }
    return newArr;
}

// 全局状态
let currentPageIndex = 0;
let answers = Array(questions.length).fill(null);

// 开始测试校验ID
function startTest(){
  const inputVal = document.getElementById("playerId").value.trim();
  if(!inputVal){
    alert("请输入你的干员ID/昵称");
    return;
  }
  playerName = inputVal;
  goPage(1);
}

// 渲染一页题目，每页5题，选项随机打乱
function renderPage(pageDomId, startIdx){
  const dom = document.getElementById(pageDomId);
  dom.innerHTML = "";
  const endIdx = Math.min(startIdx+5, questions.length);
  const total = questions.length;
  const fragment = document.createDocumentFragment();
  const wrapDiv = document.createElement("div");

  const progressText = document.createElement("div");
  progressText.className = "progress-text";
  progressText.textContent = `题目 ${startIdx+1}-${endIdx} / ${total}`;
  const progressWrap = document.createElement("div");
  progressWrap.className = "progress-wrap";
  const progressBar = document.createElement("div");
  progressBar.className = "progress-bar";
  const progressFill = document.createElement("div");
  progressFill.className = "progress-fill";
  progressFill.style.width = ((endIdx / total)*100).toFixed(1)+"%";
  progressBar.appendChild(progressFill);
  progressWrap.appendChild(progressText);
  progressWrap.appendChild(progressBar);
  wrapDiv.appendChild(progressWrap);

  for(let i=startIdx;i<endIdx;i++){
    const q = questions[i];
    // 打乱当前题目的选项
    const shuffledOptions = shuffleArray(q.options);
    
    const qDiv = document.createElement("div");
    qDiv.className = "question";
    const qTitle = document.createElement("div");
    qTitle.className = "q-title";
    qTitle.textContent = `${i+1}. ${q.question}`;
    qDiv.appendChild(qTitle);
    const optWrap = document.createElement("div");
    optWrap.className = "options";
    
    shuffledOptions.forEach((opt,optIdx)=>{
      const label = document.createElement("label");
      const input = document.createElement("input");
      input.type = "radio";
      input.name = `q_${i}`;
      // 记录原始索引，保证计分正确
      const originalIdx = q.options.indexOf(opt);
      input.value = originalIdx;
      if(answers[i] === originalIdx) input.checked = true;
      input.onchange = ()=>{ answers[i] = originalIdx; };
      label.appendChild(input);
      label.appendChild(document.createTextNode(opt.text));
      optWrap.appendChild(label);
    });
    qDiv.appendChild(optWrap);
    wrapDiv.appendChild(qDiv);
  }

  const btnGroup = document.createElement("div");
  btnGroup.className = "btn-group";
  if(startIdx>0){
    const btnPrev = document.createElement("button");
    btnPrev.className = "btn-prev";
    btnPrev.textContent = "上一页";
    btnPrev.onclick = ()=> goPage(currentPageIndex-1);
    btnGroup.appendChild(btnPrev);
  }
  if(endIdx >= questions.length){
    const btnSubmit = document.createElement("button");
    btnSubmit.className = "btn-submit";
    btnSubmit.textContent = "生成检测报告";
    btnSubmit.onclick = calcResult;
    btnGroup.appendChild(btnSubmit);
  }else{
    const btnNext = document.createElement("button");
    btnNext.className = "btn-next";
    btnNext.textContent = "下一页";
    btnNext.onclick = ()=> goPage(currentPageIndex+1);
    btnGroup.appendChild(btnNext);
  }
  wrapDiv.appendChild(btnGroup);
  fragment.appendChild(wrapDiv);
  dom.appendChild(fragment);
}

// 页面跳转
function goPage(p){
  document.querySelectorAll(".page").forEach(el=>el.classList.remove("active"));
  currentPageIndex = p;
  if(p===0){
    document.getElementById("startPage").classList.add("active");
    return;
  }else if(p===1){
    renderPage("qPage1",0);
    document.getElementById("qPage1").classList.add("active");
  }else if(p===2){
    renderPage("qPage2",5);
    document.getElementById("qPage2").classList.add("active");
  }else if(p===3){
    renderPage("qPage3",10);
    document.getElementById("qPage3").classList.add("active");
  }else if(p===4){
    renderPage("qPage4",15);
    document.getElementById("qPage4").classList.add("active");
  }
}

// 计算结果
function calcResult(){
  const unfinished = answers.some(x=>x===null);
  if(unfinished){
    alert("请完成全部题目，再生成检测报告！");
    return;
  }
  let score = {A:0,B:0,C:0,D:0};
  for(let i=0;i<questions.length;i++){
    const optIdx = answers[i];
    const opt = questions[i].options[optIdx];
    score.A += opt.score.A;
    score.B += opt.score.B;
    score.C += opt.score.C;
    score.D += opt.score.D;
  }
  // 以10分为阈值判断高低（满分20）
  const aLevel = score.A >= 10 ? "高" : "低";
  const bLevel = score.B >= 10 ? "高" : "低";
  const cLevel = score.C >= 10 ? "高" : "低";
  const dLevel = score.D >= 10 ? "高" : "低";
  const key = `A${aLevel}B${bLevel}C${cLevel}D${dLevel}`;
  const resItem = resultMap[key];
  // 渲染结果页
  document.querySelectorAll(".page").forEach(el=>el.classList.remove("active"));
  const resPage = document.getElementById("resultPage");
  resPage.classList.add("active");
  document.getElementById("resImg").innerHTML = `<img src="${resItem.img}" alt="${resItem.title}">`;
  document.getElementById("resIdLine").innerText = `那么，尊敬的${playerName}干员，您的三角洲成分为：`;
  document.getElementById("resTag").innerText = `${resItem.code} · ${resItem.title}`;
  document.getElementById("resDim").textContent = `猛攻A:${score.A} ｜ 信息B:${score.B} ｜ 兜底C:${score.C} ｜ 贪欲D:${score.D}`;
  document.getElementById("resText").innerHTML = `<p>${resItem.desc}</p>`;
}

// 复制结果文案
function copyResult(){
  const tag = document.getElementById("resTag").innerText;
  const desc = document.getElementById("resText").innerText;
  const text = `【三角洲玩家人格检测】\n干员：${playerName}\n结果：${tag}\n${desc}\n完成测试下单享500W保底`;
  navigator.clipboard.writeText(text).then(()=>{
    alert("结果已复制，请干员发给客服吧！");
  }).catch(()=>{
    alert("复制失败，请手动长按复制");
  });
}

// 重置测试
function resetTest(){
  answers = Array(questions.length).fill(null);
  playerName = "";
  document.getElementById("playerId").value = "";
  currentPageIndex = 0;
  goPage(0);
}
